CREATE HashMap patientMap

FUNCTION addPatient(id, patientData):
    patientMap.put(id, patientData)

FUNCTION getPatient(id):
    RETURN patientMap.get(id)

FUNCTION updatePatient(id, newPatientData):
    patientMap.put(id, newPatientData)

FUNCTION deletePatient(id):
    patientMap.remove(id)

MAIN:
    addPatient(101, "John Doe")
    addPatient(102, "Jane Smith")
    print getPatient(101)
    updatePatient(101, "John Doe - Updated")
    deletePatient(102)
END
