FUNCTION solveMaze(maze, x, y, solution):
    IF (x, y) is destination:
        mark solution[x][y] = 1
        RETURN true

    IF isSafe(maze, x, y):
        mark solution[x][y] = 1

        IF solveMaze(maze, x+1, y, solution):
            RETURN true
        IF solveMaze(maze, x, y+1, solution):
            RETURN true
        IF solveMaze(maze, x-1, y, solution):
            RETURN true
        IF solveMaze(maze, x, y-1, solution):
            RETURN true

        mark solution[x][y] = 0 // backtrack
        RETURN false

    RETURN false

MAIN:
    create maze grid
    create solution grid
    IF solveMaze(maze, 0, 0, solution):
        print solution
    ELSE:
        print "No Path Found"
END
