START

DEFINE CLASS Book
    ATTRIBUTES:
        title
        year
        rating

    CONSTRUCTOR(title, year, rating)
        SET this.title = title
        SET this.year = year
        SET this.rating = rating

    METHOD getDetails()
        RETURN "Title: " + title + ", Year: " + year + ", Rating: " + rating
END CLASS
