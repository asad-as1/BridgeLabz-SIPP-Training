START
DEFINE CLASS CallCenter
    ATTRIBUTE employees (double-ended queue of strings)

    METHOD addFront(employeeName)
        INSERT employeeName AT FRONT OF employees

    METHOD addLast(employeeName)
        INSERT employeeName AT END OF employees

    METHOD deleteFirst()
        REMOVE FIRST employeeName FROM employees

    METHOD deleteLast()
        REMOVE LAST employeeName FROM employees
END CLASS
END
